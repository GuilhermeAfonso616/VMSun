# --- COMANDOS ÚTEIS VMSUN ---

# 1. Execução Local via Uvicorn (Desenvolvimento)
uvicorn main:app --host 0.0.0.0 --port 8100

# 2. Atualizar código do VMSun via Git
git pull --ff-only

# 3. Gerenciamento dos Containers Docker do VMSun (Porta Web: 8100)
# Parar containers do VMSun:
docker compose down

# Fazer build dos containers (vmsun-web, vmsun-control, vmsun-camera-gateway):
docker compose build

# Subir a pilha VMSun em segundo plano:
docker compose up -d

# 4. Ver logs em tempo real:
docker compose logs -f vmsun-web

# 5. URLs de acesso:
# VMSun Web: http://localhost:8100 (ou http://<IP_DA_MAQUINA>:8100)
# VMSun Swagger Docs: http://localhost:8100/docs
